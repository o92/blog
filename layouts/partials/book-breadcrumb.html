{{/* Breadcrumb from book root to current page. Usage: {{ partial "book-breadcrumb.html" (dict "page" . "book" $book) }} */}} {{- $page := .page -}} {{- $book := .book -}} {{- if $book -}} {{- $mid := slice -}} {{- range $page.Ancestors -}} {{- if and (ne . $book) (strings.HasPrefix .Path $book.Path) -}} {{- $mid = $mid | append . -}} {{- end -}} {{- end -}} {{- $crumbs := slice $book -}} {{- range collections.Reverse $mid -}} {{- $crumbs = $crumbs | append . -}} {{- end -}} {{- if ne $page $book -}} {{- $crumbs = $crumbs | append $page -}} {{- end -}}
{{- $last := sub (len $crumbs) 1 -}} {{- range $i, $crumb := $crumbs -}} {{- if $i -}}/{{- end -}} {{- if eq $i $last -}} {{ if eq $crumb $book }}{{ partial "book-name.html" $crumb }}{{ else }}{{ $crumb.LinkTitle }}{{ end }} {{- else -}} {{ if eq $crumb $book }}{{ partial "book-name.html" $crumb }}{{ else }}{{ $crumb.LinkTitle }}{{ end }} {{- end -}} {{- end -}}
{{- end -}}
